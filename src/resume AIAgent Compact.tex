\cvproject{项目一：无人机管理云平台}{}{2025/04 -- 2026/03}{%
    \cvtech{Java、SpringBoot、Redis、MySQL、MyBatis(Plus)、WebSocket、MQTT、Vue、JavaScript、TypeScript、Cesium}
    \cvlabeled{项目概述}{在公司物联网平台基础上，利用自组网超远距离（>20km）优势新增无人机接入模块，实现空地一体化指挥调度。}
    \cvlabeled{项目实现}{%
        \begin{itemize}
            \item \textbf{无人机接入与设备管理}
              \begin{starlist}
                \item[(S)背景:]无人机经自组网硬件接入基站，链路长、断连场景复杂，现有上下线流程无法直接适配。
                \item[(T)任务:]将无人机融入现有设备管理体系，并解决服务重启后设备无法自动恢复在线的问题。
                \item[(A)动作:]通过 Redis 维护在线状态（60s TTL 心跳续期），结合设备上报的离线原因区分主动下线与异常断连；监听 Spring 启动就绪事件拉起 MQTT，订阅后向自研硬件模块逐台下发自定义重启指令，触发无人机重新上线。
                \item[(R)结果:]无人机与地面设备统一管理、共用区域权限，服务重启后自动恢复上线，无需人工值守。
              \end{starlist}

            \item \textbf{飞行控制与相机系统}
              \begin{starlist}
                \item[(S)背景:]飞行与相机指令繁多，前期为快速验证由前端直连 MQTT，逻辑分散在组件中，不利于维护与安全管控。
                \item[(T)任务:]完成主要飞行与相机指令的实现，将关键相机指令迁至后端，并补齐飞行安全校验与降落确认。
                \item[(A)动作:]前端基于 MQTT 直连实现一键起飞、指点飞行、环绕飞行、返航、急停及变焦、拍照、录像、云台重置；航线任务、框选变焦与 AIM 瞄准走后端接口；后端在飞行参数设置时做限高校验，并提供降落确认/取消接口。
                \item[(R)结果:]覆盖主要飞行与相机操控场景，限高校验与降落确认降低超限与误操作风险。
              \end{starlist}

            \item \textbf{无人机直播观看}
              \begin{starlist}
                \item[(S)背景:]硬件推流不稳定，视频流可能延迟到达甚至不可达；多指挥员需同时观看同一无人机画面。
                \item[(T)任务:]推流不可靠时仍要支持多人观看，并避免无人观看时浪费带宽。
                \item[(A)动作:]基于 Spring DeferredResult 异步确认：发起观看后释放线程，由 ZLM 流注册回调（12s 超时兜底）确认推流就绪后返回播放信息；Redis 按路维护观看用户列表做引用计数，人数归零才停止推流，缺流时重新推流。
                \item[(R)结果:]硬件延迟不再阻塞线程，多人共享同一路流互不影响，无人观看时带宽自动回收。
              \end{starlist}
        \end{itemize}
    }
}

\cvproject{项目二：WVP Copilot --- 基于 LLM 的无人机 AI Agent 智能体}{}{2025/11 -- 2026/03}{%
    \cvtech{Python、LangChain、LangGraph、SSE、MCP、FastAPI、DeepSeek API、Redis}
    \cvlabeled{项目概述}{面向消防应急指挥场景的无人机 Copilot，把自然语言需求（如“飞到陆家嘴，变焦 7x 拍照，返航”）拆解为指令序列并编排执行，以对话完成远程控制；采用 Python（Agent 编排层）+ Java（设备控制层）两层架构，通过 REST 解耦。}
    \cvlabeled{项目实现}{%
        \begin{itemize}
            \item \textbf{异步操作的状态同步与确定性故障恢复}
              \begin{starlist}
                \item[(S)背景:]LLM 推理为同步过程（秒级），无人机飞行为异步过程（分钟级），等待时长随飞行距离与通信延迟浮动。
                \item[(T)任务:]在同步推理与异步执行间建立状态同步，超时后确定性判断飞行结果，避免误判导致后续指令出错。
                \item[(A)动作:]工具函数内每秒轮询 Redis 任务状态，超时阈值按预估飞行时间动态设定；超时不直接判失败，先查 OSD 位置兜底（< 5m 判到达、无位移判硬件故障、仍在移动则延长等待）；中间进度只推前端，LLM 只拿最终状态。
                \item[(R)结果:]超时不再直接等同失败，LLM 不会被错误状态带偏下一步决策。
              \end{starlist}

            \item \textbf{多步任务链的级联故障阻断与安全隔离}
              \begin{starlist}
                \item[(S)背景:]任务链（飞行 \thickarrow 拍照 \thickarrow 返航）中任一步失败若不阻断，后续步骤会在异常状态下执行；急停须绕开 LLM。
                \item[(T)任务:]任一步失败立即中断后续执行，高风险操作有人工确认，急停通道不经过 Agent 与 LLM。
                \item[(A)动作:]Plan-Confirm-Execute 三段式：计划经用户确认卡片审核后逐步执行，工具返回失败或急停即中断后续步骤；飞行、返航另在工具层内置确认钩子；急停走独立 MQTT 通道，前端同步中断 SSE 连接。
                \item[(R)结果:]单步失败不会扩散为链式误操作，急停后前端会话与后续步骤一并终止。
              \end{starlist}

            \item \textbf{状态机驱动的子任务编排替代 LLM 时序决策}
              \begin{starlist}
                \item[(S)背景:]环绕飞行是天然状态机（进入 \thickarrow 逐角度拍照 \thickarrow 退出），但 LLM 推理 1-3s，3m/s 飞行下拍照点偏移 3-9 米。
                \item[(T)任务:]把实时控制任务的时序决策从 LLM 交给确定性状态机，同时保留 LLM 的意图理解。
                \item[(A)动作:]用 LangGraph 建模条件子图：enter\_orbit \thickarrow{} orbit\_step（按角度步进拍照，$\ge 360^\circ$ 条件边退出）\thickarrow{} exit\_orbit，执行节奏由条件边决定，LLM 只做意图路由；线性主链路仍用 AgentExecutor，避免过度设计。
                \item[(R)结果:]拍照时机由图状态转移决定，环绕时序不再受 LLM 推理延迟影响。
              \end{starlist}

            \item \textbf{MCP 外部工具链与复合指令封装}
              \begin{starlist}
                \item[(S)背景:]控制链路需接入高德地图、设备控制等外部系统，一句“录像 30 秒”即对应多条底层指令，LLM 编排易出错。
                \item[(T)任务:]集成外部工具链，把复合指令封装为业务语义，让 LLM 不直接编排硬件原语。
                \item[(A)动作:]经 MCP 接入高德地图 Server，地图工具自动注册为 LangChain Tool，地理编码多候选回传 LLM 仲裁；用 @tool 定义 16 个无人机工具，其中 record\_for\_duration 封装 start \thickarrow 倒计时 \thickarrow{} stop；SSE 流式推送工具调用。
                \item[(R)结果:]常见操作从多步收敛为单步，工具调用与执行结果对用户实时可见。
              \end{starlist}
        \end{itemize}
    }
