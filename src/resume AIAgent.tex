\cvsection{硕士毕业论文-AI算法研究及模型训练}

\cvproject{硕士毕业论文-AI算法研究及模型训练}{}{2024/06 -- 2024/12}{%
    \cvtech{Python、PyTorch、YOLOv8、imgaug、OpenCV、labelimg、AutoDL、Flask}
    \cvlabeled{项目概述}{面向复杂环境下小目标无人机检测难题，基于 YOLOv8 提出 TD-YOLO 改进算法，覆盖数据集构建、模型优化、注意力机制引入及可视化系统开发的全流程。论文获西安市科技局科技计划项目资助（22GXFW0149）。}
    \cvlabeled{项目实现}{%
        \begin{itemize}
            \item \textbf{数据集构建与增强}：基于 TIB-NET 和 UAVDrones 数据集，利用 Python imgaug 库进行多维度数据增强（随机旋转/翻转/亮度对比度调整/高斯噪声），模拟雾天、雨天、雪天等复杂天气场景，并通过背景融合技术扩充复杂背景样本，最终构建 5920 张高质量标注数据集。
            \item \textbf{TD-YOLO 算法设计}：针对小目标特征丢失和复杂环境干扰两大难点，在 YOLOv8 基础上引入三项改进——GDST-Conv 结构增强特征提取、自适应空间特征融合结构优化多尺度特征、Focal Loss 损失函数缓解类别不平衡，mAP50 较 YOLOv8 提升 1.3 个百分点。
            \item \textbf{BiFormer 注意力机制优化}：引入 BiFormer 注意力模块增强模型对无人机目标的注意力分配，有效解决 TD-YOLO 在特定天气和复杂背景下的误检测问题，mAP50 进一步提升 1.9 个百分点。
            \item \textbf{模型训练与系统开发}：基于 AutoDL 云算力平台（NVIDIA RTX 3090）完成模型训练与调优；使用 Flask 开发检测可视化系统，支持图像上传、实时推理与结果展示。
        \end{itemize}
    }
}
